# Redaction rules for diagnostics bundles.
#
# This file is the single source of these rules. nbctl carries a verbatim copy
# for its offline bundle; change both together.
#
# One rule per line, fields separated by a TAB:
#   key<TAB><regex>                  a secret-bearing key name; the value after
#                                    "key=", "key:" or "key": is replaced
#   value<TAB><regex><TAB><replace>  replace every match; $1.. refer to groups
# Regexes are Go RE2 syntax. Blank lines and lines starting with # are ignored.

key	access[_-]?key

key	cookie
value	(?i)(bearer|basic)\s+[A-Za-z0-9._~+/=-]{8,}	$1 [REDACTED]
value	eyJ[A-Za-z0-9_-]{8,}\.[A-Za-z0-9_-]{8,}\.[A-Za-z0-9_-]{8,}	[REDACTED_JWT]
value	([A-Za-z][A-Za-z0-9+.-]*://)[^/\s:@]+:[^/\s@]+@	$1[REDACTED]@
value	\b(AKIA|ASIA)[0-9A-Z]{16}\b	[REDACTED_AWS_KEY]
value	\bsk-[A-Za-z0-9_-]{16,}	[REDACTED_API_KEY]
value	(?i)((?:set-)?cookie:[ \t]*)[^\r\n]+	$1[REDACTED]
